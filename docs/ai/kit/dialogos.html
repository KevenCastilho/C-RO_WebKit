var out = document.getElementById('dl-out');
document.getElementById('dl-a').onclick = function () {
  CRO.alert('Olá, aventureiro!', { title: 'Aviso' }).then(function () {
    out.textContent = 'alert fechado';
  });
};
document.getElementById('dl-c').onclick = function () {
  CRO.confirm('Sair do jogo?', { ok: 'Sair', cancel: 'Ficar' }).then(function (v) {
    out.textContent = 'confirm: ' + v;
  });
};
document.getElementById('dl-p').onclick = function () {
  CRO.prompt('Nome do personagem:', 'Ana').then(function (v) {
    out.textContent = 'prompt: ' + JSON.stringify(v);
  });
};
document.getElementById('dl-m').onclick = function (e) {
  var r = e.target.getBoundingClientRect();
  CRO.menu([
    { label: 'Atacar', value: 'a' },
    { label: 'Falar', value: 'f' },
    '-',
    { label: 'Ignorar', value: 'i', disabled: true }
  ], r.left, r.bottom)
    .then(function (v) { out.textContent = 'menu: ' + v; });
};
Função Resolve com
CRO.alert(msg, {title, ok}) undefined
CRO.confirm(msg, {title, ok, cancel}) true / false (Esc = false)
CRO.prompt(msg, padrão, {title, ok, cancel}) texto / null (Esc ou Cancelar)
CRO.menu(itens, x, y) value do item / undefined

Com async/await

async function sair() {
  if (await CRO.confirm('Deseja sair?')) {
    const nome = await CRO.prompt('Seu nome:', 'Ana');
    if (nome !== null) await CRO.alert('Até logo, ' + nome + '!');
  }
}

Usam <dialog class="cro-modal"> com showModal() (foco preso, Esc fecha, camada superior do navegador) e uma cro-window dentro. Você pode montar o seu:

<dialog class="cro-modal">
  <section class="cro-window">

    <div class="cro-window__body cro-modal__body">Mensagem</div>
    <footer class="cro-window__footer cro-modal__actions">
      <button class="cro-btn">OK</button>
    </footer>
  </section>
</dialog>
